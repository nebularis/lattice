Version IRIs do not dereference yet. Until they do, a released version is fetched from
GitHub at its tag, named after the release and its version (foundation-v0.3.0), following
the [ontology versioning policy](ontology-versioning-policy.md#releasing-a-version). A layer's
shapes and projections are released on their own, versioned by the .version file in their
directory, and have no version IRI.

Click a file to download it under the name shown. Opening it in a new tab shows the raw
content at the tag.

<script>
// raw.githubusercontent.com allows cross-origin reads but serves text/plain, and browsers
// ignore the download attribute across origins, so fetch the file and save it locally.
document.addEventListener("click", async (event) => {
  const link = event.target.closest('a[href^="https://raw.githubusercontent.com/"]');
  if (!link || event.button !== 0 || event.metaKey || event.ctrlKey || event.shiftKey || event.altKey) return;
  event.preventDefault();
  const response = await fetch(link.href);
  if (!response.ok) { window.location.href = link.href; return; }
  const url = URL.createObjectURL(new Blob([await response.text()], { type: "text/turtle" }));
  const save = Object.assign(document.createElement("a"), { href: url, download: link.textContent });
  document.body.append(save);
  save.click();
  save.remove();
  setTimeout(() => URL.revokeObjectURL(url), 1000);
});
</script>
